```
#!/bin/bash
set -euo pipefail

# ======================================
# Input paths
# ======================================
GATK_PATH="/storage/public/home/2024050455/08.compare/Extract"
CLAIR3_PATH="/storage/public/home/2024050455/08.compare/Sample"

# ======================================
# Paired samples (one-to-one mapping)
# ======================================
samples_gatk=(
3DWF22T0036 3DWF22T0070 3DWF22T0240 3DWF22T0415 3DWF22T0549
3DWFN0018 3DWFP0355 3DWFP0398 3DWFP0413 3DWFQ0177
3DWFQ0404 3DWFQ0458 3DWFQ0459 3DWFR0017 3DWFR0281
3DWFS0069 3DWFS0123 3DWFS0137 3DWFS0202 WGWFL0149
)

samples_clair=(
sample15 sample06 sample12 sample11 sample16
sample03 sample07 sample20 sample04 sample05
sample01 sample13 sample02 sample10 sample17
sample18 sample14 sample09 sample19 sample08
)

# ======================================
# Output directories and file
# ======================================
OUTDIR="discordant_analysis"
TMPDIR="temp_files"
OUTFILE="${OUTDIR}/all_genotype_mismatches.txt"

mkdir -p "${OUTDIR}" "${TMPDIR}"

# Write header to output file
echo -e "CHROM\tPOS\tREF\tALT\tGATK_Sample\tGATK_GT\tGATK_GQ\tGATK_DP\tClair3_Sample\tClair3_GT\tClair3_GQ\tClair3_DP" \
> "${OUTFILE}"

echo "Rebuilding all_genotype_mismatches.txt ..."

# ======================================
# Main processing loop
# ======================================
for i in "${!samples_gatk[@]}"; do

    gatk_sample="${samples_gatk[$i]}"
    clair_sample="${samples_clair[$i]}"

    gatk_vcf="${GATK_PATH}/${gatk_sample}.filtered-DP20-40-GQ30.vcf.gz"
    clair_vcf="${CLAIR3_PATH}/${clair_sample}_DP10-20_GQ20.vcf.gz"

    echo "Processing ${gatk_sample} vs ${clair_sample}"

    # ----------------------------------
    # Step 1: Extract shared variant sites
    # ----------------------------------
    bcftools isec -n=2 -w1 "${gatk_vcf}" "${clair_vcf}" -Oz \
        -o "${TMPDIR}/${gatk_sample}.common.vcf.gz"
    bcftools isec -n=2 -w2 "${gatk_vcf}" "${clair_vcf}" -Oz \
        -o "${TMPDIR}/${clair_sample}.common.vcf.gz"

    bcftools index -f "${TMPDIR}/${gatk_sample}.common.vcf.gz"
    bcftools index -f "${TMPDIR}/${clair_sample}.common.vcf.gz"

    # ----------------------------------
    # Step 2: Extract genotype information
    # ----------------------------------
    # GATK:
    #   - GT and GQ from FORMAT fields
    #   - DP from INFO field (site-level depth)
    bcftools query -f '%CHROM\t%POS\t%REF\t%ALT[\t%GT\t%GQ]\t%INFO/DP\n' \
        "${TMPDIR}/${gatk_sample}.common.vcf.gz" \
        > "${TMPDIR}/${gatk_sample}.info"

    # Clair3:
    #   - GT, GQ, and DP from FORMAT fields (per-sample depth)
    bcftools query -f '%CHROM\t%POS\t%REF\t%ALT[\t%GT\t%GQ\t%DP]\n' \
        "${TMPDIR}/${clair_sample}.common.vcf.gz" \
        > "${TMPDIR}/${clair_sample}.info"

    # ----------------------------------
    # Step 3: Normalize genotypes and identify mismatches
    # ----------------------------------
    paste "${TMPDIR}/${gatk_sample}.info" "${TMPDIR}/${clair_sample}.info" | \
    awk -v gs="${gatk_sample}" -v cs="${clair_sample}" '
    BEGIN{OFS="\t"}

    # Normalize phased/unphased genotypes and allele order
    function norm_gt(gt, a, b) {
        gsub(/\|/, "/", gt)
        split(gt, arr, "/")
        a = arr[1]; b = arr[2]
        if (a > b) return b "/" a
        else return a "/" b
    }

    {
        # GATK fields
        gatk_gt = norm_gt($5)
        gatk_gq = $6
        gatk_dp = $7    # INFO/DP

        # Clair3 fields
        clair_gt = norm_gt($12)
        clair_gq = $13
        clair_dp = $14  # FORMAT/DP

        # Report true genotype discordance
        if (gatk_gt != clair_gt) {
            print $1, $2, $3, $4,
                  gs, gatk_gt, gatk_gq, gatk_dp,
                  cs, clair_gt, clair_gq, clair_dp
        }
    }' >> "${OUTFILE}"

done

echo "Done."
echo "Output file:"
echo "  ${OUTFILE}"
```

##Organized by site
```
#!/bin/bash
set -euo pipefail

# ----------------------------------
# Input / Output paths
# ----------------------------------
INDIR="discordant-analysis"
INPUT="${INDIR}/all_genotype_mismatches.txt"
OUTDIR="discordant-analysis"
TMP="${OUTDIR}/.body.tmp"
OUTPUT="${OUTDIR}/site_level_mismatch_matrix.tsv"

mkdir -p "${OUTDIR}"

# ----------------------------------
# Step 1: generate header (do NOT sort)
# ----------------------------------
awk -v OFS="\t" '
NR==1 { next }
{
    pairs[$5 "_vs_" $9] = 1
}
END {
    n = asorti(pairs, pair_list)
    printf "CHROM\tPOS\tREF\tALT"
    for (i = 1; i <= n; i++)
        printf "\t%s", pair_list[i]
    printf "\n"
}
' "${INPUT}" > "${OUTPUT}"

# ----------------------------------
# Step 2: generate body and sort by site
# ----------------------------------
awk -v OFS="\t" '
NR==1 { next }

{
    site = $1 SUBSEP $2 SUBSEP $3 SUBSEP $4
    pair = $5 "_vs_" $9

    # Cell format:
    # GATK(GT,DP,GQ)|CLAIR3(GT,DP,GQ)
    cell = "GATK(" $6 "," $8 "," $7 ")|CLAIR3(" $10 "," $12 "," $11 ")"

    sites[site] = 1
    pairs[pair] = 1
    data[site, pair] = cell
}

END {
    n = asorti(pairs, pair_list)

    for (s in sites) {
        split(s, a, SUBSEP)
        printf "%s\t%s\t%s\t%s", a[1], a[2], a[3], a[4]

        for (i = 1; i <= n; i++) {
            key = s SUBSEP pair_list[i]
            if (key in data)
                printf "\t%s", data[key]
            else
                printf "\tNA"
        }
        printf "\n"
    }
}
' "${INPUT}" | sort -k1,1 -k2,2n > "${TMP}"

# ----------------------------------
# Step 3: combine header and body
# ----------------------------------
cat "${TMP}" >> "${OUTPUT}"
rm -f "${TMP}"

echo "Done."
echo "Output file:"
echo "  ${OUTPUT}"
```

```
awk -v OFS="\t" '
NR==1 {
    total_pairs = NF - 4
    print "CHROM","POS","REF","ALT",
          "Discordant_Pair_Count","Total_Pairs","Discordance_Rate"
    next
}
{
    count = 0
    for (i = 5; i <= NF; i++) {
        if ($i != "NA")
            count++
    }
    rate = count / total_pairs
    print $1, $2, $3, $4, count, total_pairs, rate
}
' discordant-analysis/site_level_mismatch_matrix.tsv \
```
